cmake_minimum_required(VERSION 3.16)
project(antimatter_features VERSION 1.0.0 LANGUAGES C)
if(WIN32)
  message(FATAL_ERROR "The native feature socket SDK currently targets Linux and macOS.")
endif()
add_library(antimatter_features SHARED antimatter_features.c)
set_target_properties(antimatter_features PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES VERSION 1.0.0 SOVERSION 1)
target_include_directories(antimatter_features PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}> $<INSTALL_INTERFACE:include>)
target_compile_options(antimatter_features PRIVATE -Wall -Wextra -Werror)
install(TARGETS antimatter_features LIBRARY DESTINATION lib)
install(FILES antimatter_features.h DESTINATION include)
install(FILES antimatter_features.matrix DESTINATION share/antimatter/features)
